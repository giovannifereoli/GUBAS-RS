1//! Shared data structures: [`Cube`], [`Params`], and [`Initialization`].
2//!
3//! - [`Cube<T>`] — 3-D array replacing Armadillo's `cube`.  Flat row-major
4//!   storage, indexed `(l, m, n)`.
5//! - [`Params<T>`] — all read-only simulation parameters threaded through the
6//!   ODE and integrators.  Generic over [`Scalar`] so `Dual` types can replace
7//!   `f64` for auto-differentiation.
8//! - [`promote_params`] — lift `Params<f64>` → `Params<U>` for dual-number AD.
9
10use crate::math3::{Mat3, Scalar};
11
12// ── Cube ─────────────────────────────────────────────────────────────────────
13
14/// A 3-D array of size `(order+1) × (order+1) × (order+1)`.
15///
16/// Indexed as `cube[l][m][n]` using row-major flattening:
17///   flat_idx = l*(order+1)^2 + m*(order+1) + n
18///
19/// This replaces `arma::cube` from Armadillo.  Defaults to `f64` so existing
20/// callers compile without changes.
21#[derive(Clone, Debug)]
22pub struct Cube<T = f64> {
23    pub order: usize,
24    data: Vec<T>,
25}
26
27impl<T: Scalar> Cube<T> {
28    pub fn new(order: usize) -> Self {
29        let sz = order + 1;
30        Self { order, data: vec![T::zero(); sz * sz * sz] }
31    }
32
33    fn idx(&self, l: usize, m: usize, n: usize) -> usize {
34        // Without this check an index such as (0, 0, order+1) would alias
35        // (0, 1, 0) instead of failing.
36        debug_assert!(l <= self.order && m <= self.order && n <= self.order,
37                      "Cube index ({l}, {m}, {n}) exceeds order {}", self.order);
38        let sz = self.order + 1;
39        l * sz * sz + m * sz + n
40    }
41
42    #[inline]
43    pub fn get(&self, l: usize, m: usize, n: usize) -> T {
44        self.data[self.idx(l, m, n)]
45    }
46
47    #[inline]
48    pub fn set(&mut self, l: usize, m: usize, n: usize, val: T) {
49        let i = self.idx(l, m, n);
50        self.data[i] = val;
51    }
52
53    #[inline]
54    pub fn add(&mut self, l: usize, m: usize, n: usize, val: T) {
55        let i = self.idx(l, m, n);
56        self.data[i] += val;
57    }
58
59    pub fn zeros(&mut self) {
60        for v in &mut self.data { *v = T::zero(); }
61    }
62
63    /// Promote every element to a different scalar type via `NumCast`.
64    ///
65    /// Used to lift a `Cube<f64>` of constant parameters into `Cube<Dual>`
66    /// (with eps = 0) so that dual-number ODE evaluations can flow through.
67    pub fn promote<U: Scalar>(&self) -> Cube<U> {
68        let mut out = Cube::new(self.order);
69        for (i, v) in self.data.iter().enumerate() {
70            out.data[i] = <U as num_traits::NumCast>::from(v.to_f64().unwrap()).unwrap();
71        }
72        out
73    }
74
75    /// Second-moment tensor J and inertia tensor I = tr(J)·E − J of the body,
76    /// both about the origin of its body frame (kg·km²).
77    ///
78    /// J_pq = T(e_p + e_q), so J_xx = T_200, J_xy = T_110, and so on.  For
79    /// integrals taken about the centre of mass this is the inertia tensor the
80    /// rotational equations use.
81    pub fn inertia_tensor(&self) -> Mat3<T> {
82        second_moment_to_inertia(self.second_moments())
83    }
84
85    /// Second-moment tensor J_pq = T(e_p + e_q) (kg·km²).
86    pub fn second_moments(&self) -> Mat3<T> {
87        let mut j = crate::math3::zero_m::<T>();
88        for p in 0..3 {
89            for q in 0..3 {
90                let mut e = [0usize; 3];
91                e[p] += 1;
92                e[q] += 1;
93                j[p][q] = self.get(e[0], e[1], e[2]);
94            }
95        }
96        j
97    }
98}

100/// I = tr(J)·E − J, the inertia tensor of a second-moment tensor J.  Linear
101/// in J, so it maps a change ΔJ to the matching change ΔI.
102pub fn second_moment_to_inertia<T: Scalar>(j: Mat3<T>) -> Mat3<T> {
103    use crate::math3::{eye, scale_m, sub_m, trace};
104    sub_m(scale_m(trace(j), eye()), j)
105}

111/// `std::fs::read_to_string`, with the path in the error message.
112pub(crate) fn read_text(path: &str) -> std::io::Result<String> {

114        .map_err(|e| std::io::Error::new(e.kind(), format!("{path}: {e}")))

117// CSV I/O is only needed for f64 cubes (file format is plain text floats).
118impl Cube<f64> {
119    /// Save to CSV in the same layout as Armadillo's `save(..., csv_ascii)`.
120    ///
121    /// The C++ stacks slices vertically: slice 0 (`T[:,:,0]`) then slice 1
122    /// (`T[:,:,1]`) … so the CSV has `(order+1)^2` rows and `(order+1)` cols.
123    /// Each row `s*(order+1) + l`, col `m` = `T[l][m][s]`.
124    pub fn save_csv(&self, path: &str) -> std::io::Result<()> {
125        use std::io::Write;
126        let mut f = std::io::BufWriter::new(std::fs::File::create(path)?);
127        let sz = self.order + 1;
128        for s in 0..sz {
129            for l in 0..sz {
130                let vals: Vec<String> = (0..sz)
131                    .map(|m| format!("{:.16e}", self.get(l, m, s)))
132                    .collect();
133                writeln!(f, "{}", vals.join(","))?;
134            }
135        }
136        Ok(())
137    }
138
139    /// Load from the CSV format written by `save_csv` (or by the C++ tool).
140    ///
141    /// Fails with `InvalidData` on an empty file, an entry that is not a
142    /// number, or a table that is not `(order+1)^2` rows of `order+1` values.
143    pub fn load_csv(path: &str) -> std::io::Result<Self> {
144        let content = read_text(path)?;
145        let mut rows: Vec<Vec<f64>> = Vec::new();
146        for (n, line) in content.lines().enumerate() {
147            if line.trim().is_empty() { continue; }
148            let row = line.split(',')
149                .map(|s| s.trim().parse::<f64>().map_err(|e| invalid_data(
150                    path, format!("line {}: cannot parse {:?} ({e})", n + 1, s.trim()))))
151                .collect::<std::io::Result<Vec<f64>>>()?;
152            rows.push(row);
153        }
154        let sz = rows.first()
155            .map(|r| r.len())
156            .ok_or_else(|| invalid_data(path, "no data".into()))?;
157        if rows.len() != sz * sz || rows.iter().any(|r| r.len() != sz) {
158            return Err(invalid_data(path, format!(
159                "expected {} rows of {sz} values for order {}", sz * sz, sz - 1)));
160        }
161        let order = sz - 1;
162        let mut cube = Cube::new(order);
163        for s in 0..sz {
164            for l in 0..sz {
165                let row_idx = s * sz + l;
166                for m in 0..sz {
167                    cube.set(l, m, s, rows[row_idx][m]);
168                }
169            }
170        }
171        Ok(cube)
172    }
173}

177/// All read-only simulation parameters passed to the ODE and integrators.
178///
179/// Generic over `T` (defaults to `f64`) so that dual-number types can replace
180/// f64 for auto-differentiation.  Only the physics fields that appear in
181/// differentiated expressions are `T`; toggles and orbit parameters stay `f64`.
182///
183/// # Units
184/// - Length: km
185/// - Mass:   kg
186/// - Time:   s (angles in radians)
187#[derive(Clone)]
188pub struct Params<T = f64> {
189    // ── Gravity and masses ────────────────────────────────────────────────────
190    /// Gravitational constant G (km³ kg⁻¹ s⁻²).
191    pub g: T,
192    /// Reduced mass μ = Mc·Ms/(Mc+Ms) (kg).
193    pub m: T,
194    /// Secondary mass fraction ν = Ms/(Mc+Ms) (dimensionless).
195    pub nu: T,
196
197    // ── Inertia integrals ─────────────────────────────────────────────────────
198    /// T_{ijk} for primary (Didymos A) in body-fixed frame (kg·km^{i+j+k}).
199    /// `ta.get(0,0,0)` = total mass Mc.  Not pre-rotated.
200    pub ta: Cube<T>,
201    /// T_{ijk} for secondary (Dimorphos) in body-fixed frame (kg·km^{i+j+k}).
202    /// `tb.get(0,0,0)` = total mass Ms.  Not pre-rotated.
203    pub tb: Cube<T>,
204    /// Inertia tensor of the primary about its centre of mass, in its body
205    /// frame (kg·km²).  Loaded as diag(IDP.csv).  When [`Params::with_theta`]
206    /// changes degree-2 entries of `ta` it changes this tensor by the same
207    /// amount as tr(J)·E − J, J_pq = T(e_p + e_q).
208    pub ia: Mat3<T>,
209    /// Inertia tensor of the secondary about its centre of mass, in its body
210    /// frame (kg·km²).  Loaded as diag(IDS.csv); updated from `tb` like `ia`.
211    pub ib: Mat3<T>,
212
213    // ── Expansion order and combinatorial coefficients ─────────────────────────
214    /// Truncation order of the mutual gravity expansion (integer l_max).
215    pub n: usize,
216    /// T_k combinatorial weights used in the mutual potential sum (Hou 2016).
217    pub tk: Vec<Vec<f64>>,
218    /// a_k coefficients in the mutual potential expansion.
219    pub a: Vec<f64>,
220    /// b_k coefficients in the mutual potential expansion.
221    pub b: Vec<f64>,
222
223    // ── Physics toggles (0 = off, 1 = on) ─────────────────────────────────────
224    /// Enable 3rd-body flyby perturbation (hyperbolic orbit).
225    pub flyby_toggle: i32,
226    /// Enable heliocentric solar radiation pressure / tide (Keplerian approximation).
227    pub helio_toggle: i32,
228    /// Enable self-gravity spin–orbit coupling correction.
229    pub sg_toggle: i32,
230    /// Enable tidal torque dissipation (Kaula-style).
231    pub tt_toggle: i32,
232
233    // ── 3rd-body flyby orbital elements ───────────────────────────────────────
234    /// Mass of the flyby perturber (kg).
235    pub mplanet: f64,
236    /// Semi-major axis of the flyby hyperbolic orbit (km, negative for hyperbola).
237    pub a_hyp: f64,
238    /// Eccentricity of the flyby hyperbolic orbit (> 1).
239    pub e_hyp: f64,
240    /// Inclination of the flyby hyperbolic orbit (rad).
241    pub i_hyp: f64,
242    /// RAAN of the flyby orbit (rad).
243    pub raan_hyp: f64,
244    /// Argument of periapsis of the flyby orbit (rad).
245    pub om_hyp: f64,
246    /// Time of periapsis passage of the flyby orbit (s).
247    pub tau_hyp: f64,
248    /// Mean motion of the flyby hyperbolic orbit n = sqrt(G·m_planet/|a|³) (rad/s).
249    pub n_hyp: f64,
250
251    // ── Heliocentric orbit parameters ──────────────────────────────────────────
252    /// Mass of the heliocentric solar body used for tidal/radiation perturbation (kg).
253    pub msolar: f64,
254    /// Semi-major axis of the heliocentric orbit (km).
255    pub a_helio: f64,
256    /// Eccentricity of the heliocentric orbit.
257    pub e_helio: f64,
258    /// Inclination of the heliocentric orbit (rad).
259    pub i_helio: f64,
260    /// RAAN of the heliocentric orbit (rad).
261    pub raan_helio: f64,
262    /// Argument of periapsis of the heliocentric orbit (rad).
263    pub om_helio: f64,
264    /// Time of periapsis passage of the heliocentric orbit (s).
265    pub tau_helio: f64,
266    /// Mean motion of the heliocentric orbit (rad/s).
267    pub n_helio: f64,
268
269    // ── Simplified circular solar orbit (legacy) ───────────────────────────────
270    /// Heliocentric distance (AU, converted internally to km).
271    pub sol_rad: f64,
272    /// Definition of 1 AU in km (used with sol_rad).
273    pub au_def: f64,
274    /// Mean motion of the circular solar orbit (rad/s), from the nominal
275    /// masses.  A plain f64, so it is neither differentiated nor updated by
276    /// [`Params::with_theta`]; the masses change it by about M/M_sun.
277    pub mean_motion: f64,
278
279    // ── Tidal torque parameters ────────────────────────────────────────────────
280    /// Tidal Love number k₂ for the primary.
281    pub love1: f64,
282    /// Tidal Love number k₂ for the secondary.
283    pub love2: f64,
284    /// Reference radius for tidal Love number of primary (km).
285    pub refrad1: f64,
286    /// Reference radius for tidal Love number of secondary (km).
287    pub refrad2: f64,
288    /// Bulk density of primary (kg/km³).
289    pub rho_a: f64,
290    /// Bulk density of secondary (kg/km³).
291    pub rho_b: f64,
292    /// Tidal phase lag ε₁ for primary (rad).
293    pub eps1: f64,
294    /// Tidal phase lag ε₂ for secondary (rad).
295    pub eps2: f64,
296
297    // ── Modified inertia tensors for LGVI integrator ───────────────────────────
298    /// Modified inertia tensor Ĩ_A = 2(½ tr(I_A)·E − I_A) used by LGVI (kg·km²).
299    pub ida: Mat3<T>,
300    /// Modified inertia tensor Ĩ_B = 2(½ tr(I_B)·E − I_B) used by LGVI (kg·km²).
301    pub idb: Mat3<T>,
302
303    // ── Solar mass ─────────────────────────────────────────────────────────────
304    /// Mass of the Sun (kg), used to compute mean motion of the binary's heliocentric orbit.
305    pub msun: f64,
306}

308impl<T: Scalar> Params<T> {
309    /// Compute the modified inertia tensors needed by the LGVI integrator:
310    ///   IdA = 2 * (0.5*tr(IA)*I₃ − IA)
311    pub fn compute_lgvi_inertia(&mut self) {
312        fn modified<T: Scalar>(i: Mat3<T>) -> Mat3<T> {
313            use crate::math3::{eye, scale_m, sub_m, trace};
314            let two  = T::one() + T::one();
315            let half = T::one() / two;
316            let scaled_eye = scale_m(half * trace(i), eye());
317            scale_m(two, sub_m(scaled_eye, i))
318        }
319        self.ida = modified(self.ia);
320        self.idb = modified(self.ib);
321    }
322
323    /// Copy of these parameters with the integrals `idx_a` of the primary set
324    /// to `theta_a` and the integrals `idx_b` of the secondary set to
325    /// `theta_b`, and with every quantity derived from them updated to match:
326    ///
327    /// - degree-2 entries change the inertia tensors by ΔI = tr(ΔJ)·E − ΔJ,
328    ///   where ΔJ_pq is the change of T(e_p + e_q);
329    /// - degree-0 entries change the masses, so `m` and `nu` are recomputed;
330    /// - `ida`/`idb` are recomputed from the new inertia tensors.
331    ///
332    /// Without this the rotational equations would keep the nominal inertia
333    /// while the mutual potential used the new integrals.  Degree-1 entries
334    /// move the centre of mass away from the body-frame origin, which the
335    /// equations of motion assume is the centre of mass; see
336    /// [`check_theta_indices`].
337    pub fn with_theta(
338        &self,
339        idx_a: &[(usize, usize, usize)], theta_a: &[T],
340        idx_b: &[(usize, usize, usize)], theta_b: &[T],
341    ) -> Params<T> {
342        fn apply<T: Scalar>(
343            t: &mut Cube<T>, inertia: &mut Mat3<T>,
344            idx: &[(usize, usize, usize)], theta: &[T],
345        ) {
346            assert_eq!(idx.len(), theta.len(), "one value per θ index");
347            let mut dj = crate::math3::zero_m::<T>();
348            for (&(i, j, k), &value) in idx.iter().zip(theta) {
349                if i + j + k == 2 {
350                    // T(e_p + e_q) is J_pq.
351                    let (p, q) = match (i, j, k) {
352                        (2, 0, 0) => (0, 0), (0, 2, 0) => (1, 1), (0, 0, 2) => (2, 2),
353                        (1, 1, 0) => (0, 1), (1, 0, 1) => (0, 2), (0, 1, 1) => (1, 2),
354                        _ => unreachable!("i + j + k == 2"),
355                    };
356                    let delta = value - t.get(i, j, k);
357                    dj[p][q] += delta;
358                    if p != q { dj[q][p] += delta; }
359                }
360                t.set(i, j, k, value);
361            }
362            *inertia = crate::math3::add_m(*inertia, second_moment_to_inertia(dj));
363        }
364
365        let mut out = self.clone();
366        apply(&mut out.ta, &mut out.ia, idx_a, theta_a);
367        apply(&mut out.tb, &mut out.ib, idx_b, theta_b);
368        let mc = out.ta.get(0, 0, 0);
369        let ms = out.tb.get(0, 0, 0);
370        out.m  = mc * ms / (mc + ms);
371        out.nu = ms / (mc + ms);
372        out.compute_lgvi_inertia();
373        out
374    }
375}

377/// Check that a list of θ indices can be estimated with these equations of
378/// motion: every entry must fit in a cube of order `order`, and none may be of
379/// degree 1, because a first moment moves the centre of mass off the
380/// body-frame origin.
381pub fn check_theta_indices(idx: &[(usize, usize, usize)], order: usize) -> Result<(), String> {
382    for &(i, j, k) in idx {
383        if i + j + k > order {
384            return Err(format!(
385                "T_{i}{j}{k} has degree {} but the expansion order is {order}", i + j + k));
386        }
387        if i + j + k == 1 {
388            return Err(format!(
389                "T_{i}{j}{k} is a first moment; the equations of motion assume the \
390                 body-frame origin is the centre of mass, so degree 1 cannot be estimated"));
391        }
392    }
393    Ok(())
394}

398/// Lift a `Params<f64>` into `Params<U>` by promoting every physics scalar
399/// via `NumCast` (eps = 0 for dual numbers — all params are constants).
400///
401/// The inertia tensors `ida` / `idb` are recomputed from the promoted `ia`/`ib`.
402///
403/// # Example
404/// ```rust,ignore
405/// use gubas_rs::dual::Dual;
406/// use gubas_rs::types::promote_params;
407///
408/// let pd: Params<Dual> = promote_params(&params);
409/// let ode_dual = |x: [Dual; 30], t: f64| hou_ode(x, t, &pd);
410/// ```
411pub fn promote_params<U: Scalar>(p: &Params<f64>) -> Params<U> {
412    let lift = |v: f64| -> U { <U as num_traits::NumCast>::from(v).unwrap() };
413    let mut out = Params {
414        g:  lift(p.g),
415        m:  lift(p.m),
416        nu: lift(p.nu),
417        ta: p.ta.promote::<U>(),
418        tb: p.tb.promote::<U>(),
419        ia: p.ia.map(|row| row.map(lift)),
420        ib: p.ib.map(|row| row.map(lift)),
421        n:  p.n,
422        tk: p.tk.clone(),
423        a:  p.a.clone(),
424        b:  p.b.clone(),
425        flyby_toggle: p.flyby_toggle,
426        helio_toggle: p.helio_toggle,
427        sg_toggle:    p.sg_toggle,
428        tt_toggle:    p.tt_toggle,
429        mplanet:   p.mplanet,
430        a_hyp:     p.a_hyp,
431        e_hyp:     p.e_hyp,
432        i_hyp:     p.i_hyp,
433        raan_hyp:  p.raan_hyp,
434        om_hyp:    p.om_hyp,
435        tau_hyp:   p.tau_hyp,
436        n_hyp:     p.n_hyp,
437        msolar:    p.msolar,
438        a_helio:   p.a_helio,
439        e_helio:   p.e_helio,
440        i_helio:   p.i_helio,
441        raan_helio: p.raan_helio,
442        om_helio:  p.om_helio,
443        tau_helio: p.tau_helio,
444        n_helio:   p.n_helio,
445        sol_rad:   p.sol_rad,
446        au_def:    p.au_def,
447        mean_motion: p.mean_motion,
448        love1:     p.love1,
449        love2:     p.love2,
450        refrad1:   p.refrad1,
451        refrad2:   p.refrad2,
452        rho_a:     p.rho_a,
453        rho_b:     p.rho_b,
454        eps1:      p.eps1,
455        eps2:      p.eps2,
456        ida: crate::math3::zero_m::<U>(),
457        idb: crate::math3::zero_m::<U>(),
458        msun: p.msun,
459    };
460    out.compute_lgvi_inertia();
461    out
462}

509impl Initialization {
510    /// Read `ic_input.txt` from `dir`.  Fails with `InvalidData`, naming the
511    /// line, when a value is missing or cannot be parsed.
512    pub fn read(dir: &std::path::Path) -> std::io::Result<Self> {
513        struct Lines<'a> {
514            path:  String,
515            lines: std::iter::Enumerate<std::str::Lines<'a>>,
516        }
517        impl Lines<'_> {
518            fn next<V>(&mut self, name: &str) -> std::io::Result<V>
519            where V: std::str::FromStr, V::Err: std::fmt::Display {
520                let (n, line) = self.lines.next().ok_or_else(|| invalid_data(

607    /// Expansion orders `(n, n_a, n_b)` actually used.  A polyhedral body is
608    /// expanded to at least the mutual-potential order, and the mutual order
609    /// is at least each body order.
610    pub fn expansion_orders(&self) -> (usize, usize, usize) {
611        let mut order_a = self.order_a;
612        let mut order_b = self.order_b;
613        if self.a_shape == 2 { order_a = order_a.max(self.order); }
614        if self.b_shape == 2 { order_b = order_b.max(self.order); }
615        (self.order.max(order_a).max(order_b), order_a, order_b)
616    }
617}

619impl Params<f64> {
620    /// Load the parameters from the inertia files named in `ics`, relative to
621    /// `dir`.  Fails if a file is missing or malformed, or if an
622    /// inertia-integral cube is of lower order than the expansion.
623    pub fn load(dir: &std::path::Path, ics: &Initialization) -> std::io::Result<Self> {
624        use crate::inertia::load_inertia_csv;
625        let file = |name: &str| dir.join(name).to_string_lossy().into_owned();
626        let (order, _, _) = ics.expansion_orders();
627        let ta = Cube::load_csv(&file(&ics.ta_file))?;
628        let tb = Cube::load_csv(&file(&ics.tb_file))?;
629        for (name, cube) in [(&ics.ta_file, &ta), (&ics.tb_file, &tb)] {
630            if cube.order < order {
631                return Err(invalid_data(&file(name), format!(
632                    "inertia integrals of order {} but the expansion order is {order}; \
633                     regenerate them with Tgen = 1", cube.order)));
634            }
635        }
636        let ia = load_inertia_csv(&file(&ics.ia_file))?;
637        let ib = load_inertia_csv(&file(&ics.ib_file))?;
638        Ok(Self::from_initialization(ics, ta, tb, ia, ib))
639    }

641    /// Compute the inertia integrals and moments of both bodies from the
642    /// shapes in `ics` (polyhedron files relative to `dir`), save them to
643    /// the files named in `ics`, and assemble the parameters.
644    pub fn generate(dir: &std::path::Path, ics: &Initialization) -> std::io::Result<Self> {
645        use crate::inertia::{ell_mass_params_met, poly_inertia_met, poly_moi_met, save_moi_csv};
646        use crate::math3::Vec3;
647        let file = |name: &str| dir.join(name).to_string_lossy().into_owned();
648        let (order, order_a, order_b) = ics.expansion_orders();
649        let body = |label: &str, shape: i32, body_order: usize, rho: f64, abc: [f64; 3],
650                    tet: &str, vert: &str| -> std::io::Result<(Vec3, Cube)> {
651            let [a, b, c] = abc;
652            match shape {
653                0 => Ok(ell_mass_params_met(order, 0, rho, a, b, c)),
654                1 => Ok(ell_mass_params_met(order, body_order, rho, a, b, c)),
655                2 => {
656                    // A body expanded to a lower order than the mutual
657                    // potential has zero integrals above its own order.
658                    let own = poly_inertia_met(body_order, rho, &file(tet), &file(vert));
659                    let mut t = Cube::new(order);
660                    for l in 0..=own.order { for m in 0..=own.order { for n in 0..=own.order {
661                        t.set(l, m, n, own.get(l, m, n));
662                    }}}
663                    Ok((poly_moi_met(rho, &file(tet), &file(vert)), t))
664                }
665                _ => Err(std::io::Error::new(std::io::ErrorKind::InvalidInput, format!(
666                    "{label} shape flag {shape} is not 0 (sphere), 1 (ellipsoid) or 2 (polyhedron)"))),
667            }
668        };
669        let (ia, ta) = body("primary", ics.a_shape, order_a, ics.rho_a,
670                            [ics.a_a, ics.b_a, ics.c_a], &ics.tet_file_a, &ics.vert_file_a)?;
671        let (ib, tb) = body("secondary", ics.b_shape, order_b, ics.rho_b,
672                            [ics.a_b, ics.b_b, ics.c_b], &ics.tet_file_b, &ics.vert_file_b)?;
673        ta.save_csv(&file(&ics.ta_file))?;
674        tb.save_csv(&file(&ics.tb_file))?;
675        save_moi_csv(ia, &file(&ics.ia_file))?;
676        save_moi_csv(ib, &file(&ics.ib_file))?;
677        Ok(Self::from_initialization(ics, ta, tb, crate::math3::diag(ia), crate::math3::diag(ib)))
678    }

680    /// Assemble the parameters from the inputs and the inertia integrals and
681    /// tensors of both bodies.  The masses are the `T_000` entries, and the
682    /// expansion order is [`Initialization::expansion_orders`].
683    pub fn from_initialization(
684        ics: &Initialization,
685        ta: Cube<f64>, tb: Cube<f64>, ia: Mat3<f64>, ib: Mat3<f64>,
686    ) -> Self {
687        let (order, _, _) = ics.expansion_orders();
688        let mc = ta.get(0, 0, 0);
689        let ms = tb.get(0, 0, 0);
690        let mean_motion = (ics.g * (ics.msun + mc + ms)
691                           / (ics.sol_rad * ics.au_def / 1000.0).powi(3)).sqrt();
692        let mut params = Params {
693            g: ics.g,
694            m: mc * ms / (mc + ms),
695            nu: ms / (mc + ms),
696            ta, tb, ia, ib,
697            n: order,
698            tk: crate::coefficients::tk_calc(order),
699            a:  crate::coefficients::a_calc(order),
700            b:  crate::coefficients::b_calc(order),
701            flyby_toggle: ics.flyby_toggle,
702            helio_toggle: ics.helio_toggle,
703            sg_toggle:    ics.sg_toggle,
704            tt_toggle:    ics.tt_toggle,
705            mplanet:  ics.mplanet,
706            a_hyp:    ics.a_hyp, e_hyp: ics.e_hyp, i_hyp: ics.i_hyp,
707            raan_hyp: ics.raan_hyp, om_hyp: ics.om_hyp, tau_hyp: ics.tau_hyp,
708            n_hyp: (ics.g * ics.mplanet / ics.a_hyp.abs().powi(3)).sqrt(),
709            msolar:   ics.msolar,
710            a_helio:  ics.a_helio, e_helio: ics.e_helio, i_helio: ics.i_helio,
711            raan_helio: ics.raan_helio, om_helio: ics.om_helio, tau_helio: ics.tau_helio,
712            n_helio: (ics.g * ics.msolar / ics.a_helio.abs().powi(3)).sqrt(),
713            sol_rad:  ics.sol_rad,
714            au_def:   ics.au_def,
715            mean_motion,
716            love1:    ics.love1,   love2:   ics.love2,
717            refrad1:  ics.refrad1, refrad2: ics.refrad2,
718            rho_a:    ics.rho_a,   rho_b:   ics.rho_b,
719            eps1:     ics.eps1,    eps2:    ics.eps2,
720            ida: crate::math3::ZERO_M,
721            idb: crate::math3::ZERO_M,
722            msun: ics.msun,
723        };
724        params.compute_lgvi_inertia();
725        params
726    }

1012    #[test]
1013    fn with_theta_updates_inertia_from_second_moments() {
1014        let (_, _, p) = ellipsoid_params("theta_inertia");
1015        let d = 1e-3 * p.ta.get(2, 0, 0);
1016        let tol = 1e-12 * p.ia[2][2];
1017
1018        // T_200 → J_xx: ΔI = diag(0, d, d).
1019        let q = p.with_theta(&[(2, 0, 0)], &[p.ta.get(2, 0, 0) + d], &[], &[]);
1020        let expected = crate::math3::add_m(p.ia, crate::math3::diag([0.0, d, d]));
1021        for r in 0..3 { for c in 0..3 {
1022            assert!((q.ia[r][c] - expected[r][c]).abs() <= tol, "ia[{r}][{c}]");
1023        }}
1024        assert_eq!(q.ib, p.ib);
1025
1026        // T_110 → J_xy = J_yx: ΔI_xy = ΔI_yx = −d, diagonal unchanged.
1027        let q = p.with_theta(&[(1, 1, 0)], &[d], &[], &[]);
1028        assert!((q.ia[0][1] + d).abs() <= tol && (q.ia[1][0] + d).abs() <= tol);
1029        for r in 0..3 { assert!((q.ia[r][r] - p.ia[r][r]).abs() <= tol); }
1030
1031        // ida follows the new tensor.
1032        let mut r = q.clone();
1033        r.compute_lgvi_inertia();
1034        assert_eq!(q.ida, r.ida);
1035    }
